    function validate() {
      let n = document.getElementById("name").value;
      let e = document.getElementById("email").value;
      let p = document.getElementById("pass").value;

      if (n == "") {
        alert("Enter Name");
        return false;

      if (!/\S+@\S+\.\S+/.test(e)) {
        alert("Invalid Email");
        return false;
      }

        return false;
      }

      return true;
    }

      $scope.products = [
        { category: "Food", name: "Rice", price: 60, selected: false },
        { category: "Food", name: "Milk", price: 30, selected: false },
        { category: "Stationery", name: "Pen", price: 10, selected: false },
        { category: "Stationery", name: "Notebook", price: 40, selected: false },
        { category: "Electronics", name: "Mouse", price: 350, selected: false }
      ];

      $scope.cart = [];
      $scope.total = 0;
      $scope.submitted = false;

Experiment 6 Node.js, MongoDB & AngularJS
Question: Design a Command Line Application for an online super market using Node JS and MongoDB.
  a) Perform a search based on product id or name.
  b) On retrieving the results, display the product details of different brands in table format with the Price field in sorted order using AngularJS.
Part 6(a): Command Line Application using Node.js and MongoDB (Search by ID or Name)
exp6/6a_search.js
// Prerequisites:
// npm install mongodb
//
// MongoDB Setup Commands in mongosh:
// use supermarket;
// db.products.insertMany([
//   { id: 1, name: "Soap", brand: "Dove", price: 45 },
//   { id: 2, name: "Soap", brand: "Lux", price: 30 },
//   { id: 3, name: "Rice", brand: "IndiaGate", price: 90 },
//   { id: 4, name: "Rice", brand: "Daawat", price: 80 }
// ]);

const { MongoClient } = require("mongodb");
const readline = require("readline");

const url = "mongodb://localhost:27017";
const client = new MongoClient(url);

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout
});

async function main() {
  try {
    await client.connect();
    const collection = client.db("supermarket").collection("products");

    rl.question("Enter product ID or name to search: ", async (query) => {
      const isNum = !isNaN(query) && query.trim() !== "";
      const filter = isNum
        ? { id: Number(query) }
        : { name: { $regex: query, $options: "i" } };

      const results = await collection.find(filter).toArray();
      console.log("\n--- Search Results ---");
      console.table(results);

      rl.close();
      await client.close();
    });
  } catch (err) {
    console.error("Error:", err.message);
    rl.close();
  }
}

main();

Experiment 7 Express.js & AJAX
Question: To Build an
  a) API by following REST syntax and Express JS and AJAX Application.
Part 7(a) Server: REST API using Express.js
exp7/server.js
// Prerequisites:
// npm install express cors
// Run server: node server.js

const express = require("express");
const cors = require("cors");
const app = express();

app.use(cors());
app.use(express.json());

// In-memory data store
let products = [
  { id: 1, name: "Laptop", price: 55000 },
  { id: 2, name: "Mouse", price: 450 },
  { id: 3, name: "Keyboard", price: 1100 }
];

// REST API: GET all products
app.get("/api/products", (req, res) => {
  res.json(products);
});

// REST API: POST create a new product
app.post("/api/products", (req, res) => {
  const item = {
    id: products.length + 1,
    name: req.body.name,
    price: Number(req.body.price)
  };
  products.push(item);
  res.status(201).json(item);
});

app.listen(3000, () => {
  console.log("REST API Server running at http://localhost:3000");
});
Part 7(a) Client: AJAX Application consuming the REST API
exp7/client.html
<!DOCTYPE html>
<html>
<head>
  <title>REST API + AJAX Application</title>
  <style>
    body { font-family: Arial, sans-serif; }
    table { border-collapse: collapse; width: 360px; margin-top: 15px; }
    th, td { border: 1px solid #ccc; padding: 6px; text-align: left; }
    th { background: #f4f4f4; }
  </style>
</head>
<body>
  <h2>REST API and AJAX Application</h2>

  <h3>Add Product (POST via AJAX)</h3>
  <input id="name" placeholder="Product Name">
  <input id="price" type="number" placeholder="Price">
  <button onclick="addProduct()">Add</button>

  <h3>Product List (GET via AJAX)</h3>
  <button onclick="getProducts()">Refresh Products</button>

  <table>
    <thead>
      <tr>
        <th>ID</th>
        <th>Product Name</th>
        <th>Price (Rs.)</th>
      </tr>
    </thead>
    <tbody id="tableBody"></tbody>
  </table>

  <script>
    const API = "http://localhost:3000/api/products";

    // AJAX GET request
    async function getProducts() {
      try {
        const res = await fetch(API);
        const data = await res.json();
        const tbody = document.getElementById("tableBody");
        tbody.innerHTML = "";

        data.forEach(p => {
          tbody.innerHTML += `<tr><td>${p.id}</td><td>${p.name}</td><td>${p.price}</td></tr>`;
        });
      } catch (err) {
        alert("Make sure server.js is running on http://localhost:3000");
      }
    }

    // AJAX POST request
    async function addProduct() {
      const name = document.getElementById("name").value.trim();
      const price = document.getElementById("price").value.trim();
      if (!name || !price) {
        alert("Please enter both product name and price");
        return;
      }

      await fetch(API, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ name, price })
      });

      document.getElementById("name").value = "";
      document.getElementById("price").value = "";
      getProducts();
    }

    // Initial load
    getProducts();
  </script>
</body>
</html>
